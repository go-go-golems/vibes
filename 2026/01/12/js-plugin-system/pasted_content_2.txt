Below is a complete **browser-side** sketch that:

* runs untrusted plugin JS inside **QuickJS (WASM)** using `quickjs-emscripten` (loaded in a **Web Worker**) ([jsDelivr][1])
* exposes **only** a single host capability to the plugin: `__hostDispatch(actionJson)` via `vm.newFunction` + `vm.setProp` ([jsDelivr][2])
* lets the plugin return a **data-only UI tree** (widget DSL), which React interprets into real components.

---

## 1) `pluginSandbox.worker.ts` (QuickJS in a Worker)

```ts
// pluginSandbox.worker.ts
// Create the worker with { type: "module" } so this ESM import works.
import { getQuickJS } from "https://esm.sh/quickjs-emscripten@0.25.0";

type RpcRequest =
  | { id: number; type: "loadPlugin"; pluginId: string; code: string }
  | { id: number; type: "render"; pluginId: string; widgetId: string; state: any }
  | { id: number; type: "event"; pluginId: string; widgetId: string; handler: string; event: any; state: any };

type RpcResponse =
  | { id: number; ok: true; result: any }
  | { id: number; ok: false; error: any };

// Messages we emit to the main thread when the plugin calls dispatch()
type DispatchMessage = { type: "dispatch"; pluginId: string; actionJson: string };

const BOOTSTRAP = `
  // Data-only widget DSL: everything returned is JSON-serializable.
  (function () {
    function node(kind, props, children) {
      return { kind, props: props || {}, children: children || [] };
    }

    globalThis.ui = {
      panel(children, props) { return node("panel", props, children); },
      row(children, props) { return node("row", props, children); },
      text(text, props) { return { kind: "text", props: props || {}, text: String(text) }; },
      badge(text, props) { return { kind: "badge", props: props || {}, text: String(text) }; },

      button(label, opts) {
        opts = opts || {};
        return node("button", { label: String(label), onClick: opts.onClick }, []);
      },

      input(value, opts) {
        opts = opts || {};
        return node("input", {
          value: value == null ? "" : String(value),
          placeholder: opts.placeholder || "",
          onChange: opts.onChange
        }, []);
      },
    };

    globalThis.createActions = function(namespace, names) {
      const out = {};
      for (const n of names) {
        out[n] = function(payload) {
          return { type: namespace + "/" + n, payload };
        };
      }
      return out;
    };

    // definePlugin stores the plugin in globalThis.__plugin
    globalThis.__plugin = null;
    globalThis.definePlugin = function(defOrFactory) {
      globalThis.__plugin =
        (typeof defOrFactory === "function")
          ? defOrFactory({ ui: globalThis.ui, createActions: globalThis.createActions })
          : defOrFactory;
    };
  })();
`;

type PluginVm = {
  vm: any; // QuickJSContext
  deadlineMs: number;
};

const plugins = new Map<string, PluginVm>();
let QuickJS: any | null = null;

async function ensureQuickJS() {
  if (!QuickJS) QuickJS = await getQuickJS();
  return QuickJS;
}

function unwrap(vm: any, result: any) {
  if (result.error) {
    const err = vm.dump(result.error);
    result.error.dispose();
    throw err;
  }
  return result.value; // QuickJSHandle (caller disposes)
}

function setGlobalString(vm: any, name: string, value: string) {
  const h = vm.newString(value);
  vm.setProp(vm.global, name, h);
  h.dispose();
}

function evalJson(vm: any, code: string, filename?: string) {
  const h = unwrap(vm, vm.evalCode(code, filename));
  const s = vm.getString(h);
  h.dispose();
  return JSON.parse(s);
}

function createPluginVm(pluginId: string) {
  const vm = QuickJS!.newContext();

  // Interrupt handler: per-plugin deadline to prevent infinite loops from hanging the worker.
  let deadlineMs = Number.POSITIVE_INFINITY;
  vm.runtime.setInterruptHandler(() => Date.now() > deadlineMs);

  // Expose ONLY dispatch back to host (main thread).
  const hostDispatch = vm.newFunction("__hostDispatch", (actionJsonHandle: any) => {
    const actionJson = vm.getString(actionJsonHandle);
    const msg: DispatchMessage = { type: "dispatch", pluginId, actionJson };
    // @ts-ignore
    postMessage(msg);
    return vm.undefined;
  });
  vm.setProp(vm.global, "__hostDispatch", hostDispatch);
  hostDispatch.dispose();

  // Install widget DSL + definePlugin
  unwrap(vm, vm.evalCode(BOOTSTRAP, "bootstrap.js")).dispose();

  const entry: PluginVm = {
    vm,
    get deadlineMs() {
      return deadlineMs;
    },
    set deadlineMs(v: number) {
      deadlineMs = v;
    },
  } as any;

  plugins.set(pluginId, entry);
  return entry;
}

function getOrCreate(pluginId: string) {
  if (plugins.has(pluginId)) return plugins.get(pluginId)!;
  return createPluginVm(pluginId);
}

function withDeadline(p: PluginVm, ms: number, fn: () => any) {
  p.deadlineMs = Date.now() + ms;
  try {
    return fn();
  } finally {
    p.deadlineMs = Number.POSITIVE_INFINITY;
  }
}

// --- RPC handlers ---

async function handleLoadPlugin(msg: Extract<RpcRequest, { type: "loadPlugin" }>) {
  await ensureQuickJS();
  const p = getOrCreate(msg.pluginId);
  const { vm } = p;

  return withDeadline(p, 50, () => {
    // Evaluate plugin code; it calls definePlugin(...)
    unwrap(vm, vm.evalCode(msg.code, `${msg.pluginId}.plugin.js`)).dispose();

    // Read plugin metadata (data-only)
    const meta = evalJson(
      vm,
      `
      JSON.stringify({
        id: __plugin && __plugin.id,
        title: __plugin && __plugin.title,
        widgets: __plugin && __plugin.widgets ? Object.keys(__plugin.widgets) : []
      })
      `
    );
    return meta;
  });
}

async function handleRender(msg: Extract<RpcRequest, { type: "render" }>) {
  const p = getOrCreate(msg.pluginId);
  const { vm } = p;

  return withDeadline(p, 50, () => {
    setGlobalString(vm, "__WIDGET_ID", msg.widgetId);
    setGlobalString(vm, "__STATE_JSON", JSON.stringify(msg.state ?? null));

    const tree = evalJson(
      vm,
      `
      (function(){
        const state = JSON.parse(__STATE_JSON);
        const w = __plugin.widgets[__WIDGET_ID];
        const tree = w.render({ state: state });
        return JSON.stringify(tree);
      })()
      `
    );

    return tree;
  });
}

async function handleEvent(msg: Extract<RpcRequest, { type: "event" }>) {
  const p = getOrCreate(msg.pluginId);
  const { vm } = p;

  return withDeadline(p, 50, () => {
    setGlobalString(vm, "__WIDGET_ID", msg.widgetId);
    setGlobalString(vm, "__HANDLER", msg.handler);
    setGlobalString(vm, "__STATE_JSON", JSON.stringify(msg.state ?? null));
    setGlobalString(vm, "__EVENT_JSON", JSON.stringify(msg.event ?? null));

    // Handler gets { dispatch, state, event } only
    unwrap(
      vm,
      vm.evalCode(
        `
        (function(){
          const state = JSON.parse(__STATE_JSON);
          const event = JSON.parse(__EVENT_JSON);

          const widget = __plugin.widgets[__WIDGET_ID];
          const fn = widget.handlers[__HANDLER];

          const dispatch = (action) => __hostDispatch(JSON.stringify(action));

          fn({ dispatch, state, event });
          return 0;
        })()
        `,
        "handler.js"
      )
    ).dispose();

    return null;
  });
}

// @ts-ignore
self.onmessage = async (e: MessageEvent<RpcRequest>) => {
  const msg = e.data;
  let resp: RpcResponse;

  try {
    let result: any;
    if (msg.type === "loadPlugin") result = await handleLoadPlugin(msg);
    else if (msg.type === "render") result = await handleRender(msg);
    else result = await handleEvent(msg);

    resp = { id: msg.id, ok: true, result };
  } catch (error) {
    resp = { id: msg.id, ok: false, error };
  }

  // @ts-ignore
  postMessage(resp);
};
```

## 2) `pluginSandboxClient.ts` (main thread RPC + Redux dispatch bridge)

```ts
// pluginSandboxClient.ts
import type { Store } from "@reduxjs/toolkit";

type RpcResponse = { id: number; ok: true; result: any } | { id: number; ok: false; error: any };
type DispatchMessage = { type: "dispatch"; pluginId: string; actionJson: string };

export type PluginMeta = { id: string; title?: string; widgets: string[] };

export class PluginSandboxClient {
  private worker: Worker;
  private nextId = 1;
  private pending = new Map<number, { resolve: (v: any) => void; reject: (e: any) => void }>();
  private allow: (pluginId: string, action: any) => boolean;

  constructor(args: {
    store: Store;
    workerUrl: URL;
    allowDispatch?: (pluginId: string, action: any) => boolean;
  }) {
    const { store, workerUrl } = args;

    this.allow =
      args.allowDispatch ??
      ((pluginId, action) => typeof action?.type === "string" && action.type.startsWith(`plugin.${pluginId}/`));

    this.worker = new Worker(workerUrl, { type: "module" });

    this.worker.onmessage = (e: MessageEvent<any>) => {
      const data = e.data as RpcResponse | DispatchMessage;

      if ((data as DispatchMessage).type === "dispatch") {
        const { pluginId, actionJson } = data as DispatchMessage;
        const action = JSON.parse(actionJson);

        if (this.allow(pluginId, action)) {
          store.dispatch(action);
        }
        return;
      }

      const resp = data as RpcResponse;
      const p = this.pending.get(resp.id);
      if (!p) return;

      this.pending.delete(resp.id);

      if (resp.ok) p.resolve(resp.result);
      else p.reject(resp.error);
    };
  }

  private call(type: string, payload: any) {
    const id = this.nextId++;
    const msg = { id, type, ...payload };
    const p = new Promise<any>((resolve, reject) => this.pending.set(id, { resolve, reject }));
    this.worker.postMessage(msg);
    return p;
  }

  loadPlugin(pluginId: string, code: string): Promise<PluginMeta> {
    return this.call("loadPlugin", { pluginId, code });
  }

  render(pluginId: string, widgetId: string, state: any): Promise<any> {
    return this.call("render", { pluginId, widgetId, state });
  }

  event(pluginId: string, widgetId: string, handler: string, event: any, state: any): Promise<void> {
    return this.call("event", { pluginId, widgetId, handler, event, state });
  }
}
```

## 3) `uiTypes.ts` + `renderUi.tsx` (interpret UI tree → React elements)

```ts
// uiTypes.ts
export type UIEventRef = { handler: string; args?: any };

export type UINode =
  | { kind: "panel" | "row"; props?: any; children?: UINode[] }
  | { kind: "text" | "badge"; props?: any; text: string }
  | { kind: "button"; props: { label: string; onClick?: UIEventRef } }
  | { kind: "input"; props: { value: string; placeholder?: string; onChange?: UIEventRef } };
```

```tsx
// renderUi.tsx
import React from "react";
import type { UINode, UIEventRef } from "./uiTypes";

export function renderUINode(
  node: UINode,
  onEvent: (ref: UIEventRef, eventPayload?: any) => void
): React.ReactNode {
  if (!node) return null;

  switch (node.kind) {
    case "panel":
      return (
        <div style={{ border: "1px solid #ddd", borderRadius: 8, padding: 12, marginBottom: 12 }}>
          {(node.children ?? []).map((c, i) => (
            <React.Fragment key={i}>{renderUINode(c, onEvent)}</React.Fragment>
          ))}
        </div>
      );

    case "row":
      return (
        <div style={{ display: "flex", gap: 8, alignItems: "center", marginBottom: 8 }}>
          {(node.children ?? []).map((c, i) => (
            <React.Fragment key={i}>{renderUINode(c, onEvent)}</React.Fragment>
          ))}
        </div>
      );

    case "text":
      return <div style={{ marginBottom: 8 }}>{node.text}</div>;

    case "badge":
      return (
        <span style={{ padding: "2px 8px", borderRadius: 999, background: "#eee", marginRight: 8 }}>
          {node.text}
        </span>
      );

    case "button": {
      const { label, onClick } = node.props;
      return (
        <button
          onClick={() => onClick && onEvent(onClick, onClick.args)}
          style={{ padding: "6px 10px", borderRadius: 6, border: "1px solid #ccc", background: "white" }}
        >
          {label}
        </button>
      );
    }

    case "input": {
      const { value, placeholder, onChange } = node.props;
      return (
        <input
          value={value}
          placeholder={placeholder}
          onChange={(e) => onChange && onEvent(onChange, { value: e.target.value })}
          style={{ padding: "6px 10px", borderRadius: 6, border: "1px solid #ccc" }}
        />
      );
    }
  }
}
```

## 4) `PluginWidget.tsx` (React component that renders + routes events back into QuickJS)

```tsx
// PluginWidget.tsx
import React from "react";
import { useSelector } from "react-redux";
import type { RootState } from "./store";
import type { UINode, UIEventRef } from "./uiTypes";
import { renderUINode } from "./renderUi";
import { PluginSandboxClient } from "./pluginSandboxClient";

export function PluginWidget(props: {
  sandbox: PluginSandboxClient;
  pluginId: string;
  widgetId: string;
}) {
  const { sandbox, pluginId, widgetId } = props;

  // For the demo, pass the whole store. In a real app you'd pass a filtered view.
  const state = useSelector((s: RootState) => s);

  const [tree, setTree] = React.useState<UINode | null>(null);

  React.useEffect(() => {
    let alive = true;
    sandbox.render(pluginId, widgetId, state).then((t) => alive && setTree(t));
    return () => {
      alive = false;
    };
  }, [sandbox, pluginId, widgetId, state]);

  const onEvent = React.useCallback(
    (ref: UIEventRef, eventPayload?: any) => {
      sandbox.event(pluginId, widgetId, ref.handler, eventPayload, state);
    },
    [sandbox, pluginId, widgetId, state]
  );

  if (!tree) return <div>Loading…</div>;
  return <>{renderUINode(tree, onEvent)}</>;
}
```

## 5) Minimal Redux store + demo app

```ts
// store.ts
import { configureStore, createSlice } from "@reduxjs/toolkit";

const scenesSlice = createSlice({
  name: "scenes",
  initialState: { lastScene: "—" },
  reducers: {},
  extraReducers: (builder) => {
    builder.addMatcher(
      (action) => action.type === "plugin.scenes/sceneRequested",
      (state, action: any) => {
        state.lastScene = action.payload?.sceneId ?? "—";
      }
    );
  },
});

export const store = configureStore({
  reducer: {
    scenes: scenesSlice.reducer,
  },
});

export type RootState = ReturnType<typeof store.getState>;
```

```tsx
// App.tsx
import React from "react";
import { Provider } from "react-redux";
import { store } from "./store";
import { PluginSandboxClient } from "./pluginSandboxClient";
import { PluginWidget } from "./PluginWidget";

// Example plugin code the user loads (string).
const pluginCode = `
definePlugin(({ ui, createActions }) => {
  const actions = createActions("plugin.scenes", ["sceneRequested"]);

  return {
    id: "scenes",
    title: "Scene Controls",
    widgets: {
      ScenePad: {
        title: "Living Room",
        render({ state }) {
          return ui.panel([
            ui.text("Last scene: " + state.scenes.lastScene),
            ui.row([
              ui.button("Cozy",   { onClick: { handler: "cozy" } }),
              ui.button("Bright", { onClick: { handler: "bright" } }),
            ])
          ]);
        },
        handlers: {
          cozy({ dispatch }) {
            dispatch(actions.sceneRequested({ sceneId: "cozy" }));
          },
          bright({ dispatch }) {
            dispatch(actions.sceneRequested({ sceneId: "bright" }));
          },
        },
      },
    },
  };
});
`;

export function App() {
  const sandboxRef = React.useRef<PluginSandboxClient | null>(null);
  const [ready, setReady] = React.useState(false);

  React.useEffect(() => {
    const sandbox = new PluginSandboxClient({
      store,
      workerUrl: new URL("./pluginSandbox.worker.ts", import.meta.url),
      allowDispatch: (pluginId, action) => action?.type?.startsWith(`plugin.${pluginId}/`),
    });
    sandboxRef.current = sandbox;

    sandbox.loadPlugin("scenes", pluginCode).then(() => setReady(true));
  }, []);

  return (
    <Provider store={store}>
      <div style={{ padding: 16 }}>
        <h2>OpsDeck demo</h2>
        {ready && sandboxRef.current ? (
          <PluginWidget sandbox={sandboxRef.current} pluginId="scenes" widgetId="ScenePad" />
        ) : (
          <div>Booting sandbox…</div>
        )}
      </div>
    </Provider>
  );
}
```

### Why this matches your constraint (“store access but no more”)

* Plugin code can only influence the app by calling `dispatch(action)` inside handlers.
* The only host bridge is `__hostDispatch(JSON.stringify(action))` (everything else is pure JS inside QuickJS).
* React rendering happens from a **data-only** widget tree returned by `render()`, so plugins never get to run React/DOM code directly.

If you want the next step, say “add a small **allowed action catalog** and generate *typed* action creators for plugins” (so plugins can’t even invent action types).
