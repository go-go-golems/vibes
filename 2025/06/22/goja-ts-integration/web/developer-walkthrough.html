Developer Walkthrough: TypeScript in Go with Goja

This walkthrough will guide you through the process of setting up and running a TypeScript-in-Go integration using Goja. We'll build a simple application that demonstrates how to expose Go functions to TypeScript, compile and bundle TypeScript code, and run it within a Go application.

Prerequisites

Before starting, ensure you have:

  • Go 1.18+ installed
  • Node.js and npm installed
  • Basic knowledge of Go and TypeScript

Step 1: Create Project Structure

First, let's create our project structure:

mkdir -p myvm/cmd/server myvm/internal/handlers myvm/web myvm/build
cd myvm

Initialize the Go module:

go mod init github.com/example/myvm

Step 2: Implement Go Handler

Create a Go handler that we'll expose to TypeScript:

cat > internal/handlers/user.go << 'EOF'
package handlers

import (
    "errors"
    "time"
)

// User represents a user in the system
type User struct {
    ID        string    `json:"id"`
    Username  string    `json:"username"`
    Email     string    `json:"email"`
    CreatedAt time.Time `json:"createdAt"`
    IsActive  bool      `json:"isActive"`
}

// UserCreateParams contains parameters for creating a new user
type UserCreateParams struct {
    Username string `json:"username"`
    Email    string `json:"email"`
}

// CreateUser creates a new user in the system
// This function will be exposed to TypeScript
func CreateUser(params UserCreateParams) (*User, error) {
    // Validate input
    if params.Username == "" {
        return nil, errors.New("username is required")
    }
    if params.Email == "" {
        return nil, errors.New("email is required")
    }

    // In a real app, we would save to database here
    user := &User{
        ID:        generateID(),
        Username:  params.Username,
        Email:     params.Email,
        CreatedAt: time.Now(),
        IsActive:  true,
    }

    return user, nil
}

// Helper function to generate a simple ID
// In production, use a proper UUID library
func generateID() string {
    return "user_" + time.Now().Format("20060102150405")
}
EOF

Step 3: Set Up TypeScript Project

Initialize the TypeScript project:

mkdir -p web/types
cd web
npm init -y
npm install --save-dev typescript esbuild

Create TypeScript configuration:

cat > tsconfig.json << 'EOF'
{
  "compilerOptions": {
    "target": "ES2019",
    "module": "CommonJS",
    "strict": true,
    "esModuleInterop": true,
    "skipLibCheck": true,
    "forceConsistentCasingInFileNames": true,
    "outDir": "./dist",
    "declaration": true,
    "baseUrl": ".",
    "paths": {
      "@go/*": ["types/*"]
    }
  },
  "include": ["*.ts", "**/*.ts"],
  "exclude": ["node_modules", "dist"]
}
EOF

Create TypeScript entry point:

cat > app.ts << 'EOF'
// app.ts - Main TypeScript entry point for our application

// Import types from Go (these will be generated later)
// import { User, UserCreateParams } from '@go/handlers';

/**
 * This is our main application entry point that will be executed by Goja
 * 
 * In a real application, this would contain business logic that calls into
 * the Go-exposed APIs.
 */

// Declare the Go functions we expect to be available
// These will be properly typed once we generate the TypeScript definitions
declare function CreateUser(params: any): any;

// Example usage of the Go-exposed function
function main() {
  try {
    console.log("TypeScript application starting...");
    
    // Call the Go function exposed to our VM
    const user = CreateUser({
      username: "testuser",
      email: "test@example.com"
    });
    
    console.log("User created successfully:");
    console.log(`ID: ${user.id}`);
    console.log(`Username: ${user.username}`);
    console.log(`Email: ${user.email}`);
    console.log(`Created at: ${user.createdAt}`);
    console.log(`Active: ${user.isActive}`);
    
    return { success: true, user };
  } catch (error) {
    console.error("Error creating user:", error);
    return { success: false, error: String(error) };
  }
}

// Execute the main function
const result = main();
console.log("Execution result:", result);

// Export the result for potential use by the Go host
module.exports = { result };
EOF

Step 4: Create TypeScript Type Definitions

Since we're not using the automated type generation in this walkthrough, let's manually create the TypeScript definitions:

cat > types/handlers.ts << 'EOF'
// Generated TypeScript definitions for Go types
// This would normally be generated by tygo

export interface User {
    id: string;
    username: string;
    email: string;
    createdAt: string;
    isActive: boolean;
}

export interface UserCreateParams {
    username: string;
    email: string;
}

// Function signature for CreateUser
export declare function CreateUser(params: UserCreateParams): User | Error;
EOF

Step 5: Set Up Build Pipeline

Create the build pipeline:

cd ..
cat > build/generate.go << 'EOF'
//go:build generate
package main

// 1️⃣ Type-check (no JS emitted)
//go:generate npx tsc --project ../web/tsconfig.json --noEmit
// 2️⃣ Produce CommonJS bundle that Goja can load
//go:generate npx esbuild ../web/app.ts --bundle --format=cjs --platform=node --target=es2019 --sourcemap=inline --outfile=../web/app.js

func main() {} // never executed
EOF

Step 6: Bundle TypeScript to JavaScript

Run the build pipeline:

cd web
npx esbuild app.ts --bundle --format=cjs --platform=node --target=es2019 --sourcemap=inline --outfile=app.js
cd ..

Step 7: Create Go Server with Goja Integration

Install required Go packages:

go get github.com/dop251/goja github.com/dop251/goja_nodejs/require github.com/rs/zerolog

Create the server:

mkdir -p cmd/server/assets
cp web/app.js cmd/server/assets/

Now create the main Go file:

cat > cmd/server/main.go << 'EOF'
package main

import (
    "embed"
    "fmt"
    "github.com/dop251/goja"
    "github.com/dop251/goja_nodejs/require"
    "github.com/example/myvm/internal/handlers"
    "github.com/rs/zerolog"
    "github.com/rs/zerolog/log"
    "os"
)

//go:embed assets/app.js
var jsBundle embed.FS

// setupLogger configures zerolog with caller information for debugging
func setupLogger() {
    // Configure zerolog to include caller information
    log.Logger = log.Output(zerolog.ConsoleWriter{Out: os.Stderr}).
        With().
        Caller().
        Logger()

    // Set global log level
    zerolog.SetGlobalLevel(zerolog.DebugLevel)
}

func main() {
    // Set up zerolog with caller info
    setupLogger()
    log.Info().Msg("Starting application")

    // Create a new JavaScript VM
    vm := goja.New()
    log.Debug().Msg("JavaScript VM created")

    // Set up a Registry with our own source loader
    reg := require.NewRegistry(require.WithLoader(func(path string) ([]byte, error) {
        log.Debug().Str("path", path).Msg("Loading module")
        
        // Handle direct app.js request or node_modules/app.js request
        var fullPath string
        if path == "app.js" || path == "node_modules/app.js" {
            fullPath = "assets/app.js"
        } else {
            // For other modules, use the standard path
            fullPath = path
        }
        
        data, err := jsBundle.ReadFile(fullPath)
        if err != nil {
            log.Error().
                Err(err).
                Str("requestedPath", path).
                Str("fullPath", fullPath).
                Msg("Failed to load module")
            return nil, err
        }
        return data, nil
    }))

    // Hook Node polyfills (fs, path, etc.)
    reg.Enable(vm)
    log.Debug().Msg("Node.js polyfills enabled")

    // Define console object by binding Go functions
    console := vm.NewObject()
    if err := console.Set("log", func(call goja.FunctionCall) goja.Value {
        args := make([]interface{}, len(call.Arguments))
        for i, arg := range call.Arguments {
            args[i] = arg.Export()
        }
        fmt.Println(args...)
        return goja.Undefined()
    }); err != nil {
        log.Fatal().Err(err).Msg("Failed to set console.log")
    }
    
    if err := console.Set("error", func(call goja.FunctionCall) goja.Value {
        args := make([]interface{}, len(call.Arguments))
        for i, arg := range call.Arguments {
            args[i] = arg.Export()
        }
        fmt.Println("ERROR:", args)
        return goja.Undefined()
    }); err != nil {
        log.Fatal().Err(err).Msg("Failed to set console.error")
    }
    
    if err := vm.Set("console", console); err != nil {
        log.Fatal().Err(err).Msg("Failed to define console object")
    }

    // Expose Go handlers to JavaScript
    if err := vm.Set("CreateUser", handlers.CreateUser); err != nil {
        log.Fatal().
            Err(err).
            Msg("Failed to expose CreateUser function to JavaScript")
    }
    log.Debug().Msg("Go handlers exposed to JavaScript")

    // Execute the JavaScript bundle
    log.Info().Msg("Executing JavaScript bundle")
    
    // Execute with error handling
    var result goja.Value
    func() {
        defer func() {
            if r := recover(); r != nil {
                log.Error().
                    Interface("panic", r).
                    Msg("JavaScript execution failed with panic")
            }
        }()
        
        var err error
        result, err = vm.RunString(`
            try {
                // Call the Go function exposed to our VM
                const user = CreateUser({
                    username: "testuser",
                    email: "test@example.com"
                });
                
                console.log("User created successfully:");
                console.log("ID: " + user.id);
                console.log("Username: " + user.username);
                console.log("Email: " + user.email);
                console.log("Created at: " + user.createdAt);
                console.log("Active: " + user.isActive);
                
                ({ success: true, user: user });
            } catch (error) {
                console.error("Error creating user:", error);
                ({ success: false, error: String(error) });
            }
        `)
        if err != nil {
            log.Error().Err(err).Msg("Failed to execute JavaScript")
        }
    }()

    // Print the result
    fmt.Println("Execution completed with result:", result)
    log.Info().
        Interface("result", result).
        Msg("JavaScript execution completed")
}
EOF

Step 9: Test Error Handling

Let's modify the JavaScript to trigger an error:

cat > cmd/server/test-error.go << 'EOF'
package main

import (
    "fmt"
    "github.com/dop251/goja"
    "github.com/example/myvm/internal/handlers"
    "github.com/rs/zerolog/log"
)

func testError() {
    // Create a new JavaScript VM
    vm := goja.New()
    
    // Define console object
    console := vm.NewObject()
    console.Set("log", func(call goja.FunctionCall) goja.Value {
        args := make([]interface{}, len(call.Arguments))
        for i, arg := range call.Arguments {
            args[i] = arg.Export()
        }
        fmt.Println(args...)
        return goja.Undefined()
    })
    console.Set("error", func(call goja.FunctionCall) goja.Value {
        args := make([]interface{}, len(call.Arguments))
        for i, arg := range call.Arguments {
            args[i] = arg.Export()
        }
        fmt.Println("ERROR:", args)
        return goja.Undefined()
    })
    vm.Set("console", console)
    
    // Expose Go handlers
    vm.Set("CreateUser", handlers.CreateUser)
    
    // Execute with missing username to trigger error
    result, err := vm.RunString(`
        try {
            const user = CreateUser({
                email: "test@example.com"
                // username is missing
            });
            
            console.log("User created successfully:", user);
            ({ success: true, user: user });
        } catch (error) {
            console.error("Error creating user:", error);
            ({ success: false, error: String(error) });
        }
    `)
    
    if err != nil {
        log.Error().Err(err).Msg("Failed to execute JavaScript")
    }
    
    fmt.Println("Error test result:", result)
}
EOF

To run this test, add a call to testError() in the main function and run the application again.

Step 10: Add Development Mode with Hot Reload

For a better development experience, let's add hot reload functionality:

go get github.com/fsnotify/fsnotify

Then add this function to your main.go:

// isDevelopmentMode returns true if the app is running in development mode
func isDevelopmentMode() bool {
    return os.Getenv("APP_ENV") == "development"
}

// watchForChanges watches for file changes and triggers hot reload
func watchForChanges(vm *goja.Runtime) error {
    watcher, err := fsnotify.NewWatcher()
    if err != nil {
        return err
    }
    defer watcher.Close()

    // Watch TypeScript files
    err = watcher.Add("web")
    if err != nil {
        return err
    }

    log.Info().Msg("Watching for file changes (hot reload enabled)")

    for {
        select {
        case event, ok := <-watcher.Events:
            if !ok {
                return nil
            }
            
            // Only react to TypeScript file changes
            if !strings.HasSuffix(event.Name, ".ts") {
                continue
            }
            
            if event.Op&(fsnotify.Write|fsnotify.Create) != 0 {
                log.Info().Str("file", event.Name).Msg("File changed, rebuilding")
                
                // Rebuild TypeScript
                cmd := exec.Command("npx", "esbuild", 
                    "app.ts", 
                    "--bundle", 
                    "--format=cjs", 
                    "--platform=node", 
                    "--target=es2019", 
                    "--sourcemap=inline",
                    "--outfile=app.js")
                
                cmd.Dir = "web"
                cmd.Stdout = os.Stdout
                cmd.Stderr = os.Stderr
                
                if err := cmd.Run(); err != nil {
                    log.Error().Err(err).Msg("Failed to rebuild TypeScript")
                    continue
                }
                
                // Copy the new bundle to the assets directory
                data, err := os.ReadFile("web/app.js")
                if err != nil {
                    log.Error().Err(err).Msg("Failed to read new bundle")
                    continue
                }
                
                if err := os.WriteFile("cmd/server/assets/app.js", data, 0644); err != nil {
                    log.Error().Err(err).Msg("Failed to update bundle")
                    continue
                }
                
                log.Info().Msg("Bundle updated, reloading...")
                
                // Execute the updated code
                func() {
                    defer func() {
                        if r := recover(); r != nil {
                            log.Error().
                                Interface("panic", r).
                                Msg("JavaScript execution failed with panic")
                        }
                    }()
                    
                    // Re-execute the JavaScript
                    result, err := vm.RunString(`
                        try {
                            // Call the Go function exposed to our VM
                            const user = CreateUser({
                                username: "testuser",
                                email: "test@example.com"
                            });
                            
                            console.log("User created successfully (hot reload):");
                            console.log("ID: " + user.id);
                            console.log("Username: " + user.username);
                            console.log("Email: " + user.email);
                            console.log("Created at: " + user.createdAt);
                            console.log("Active: " + user.isActive);
                            
                            ({ success: true, user: user });
                        } catch (error) {
                            console.error("Error creating user:", error);
                            ({ success: false, error: String(error) });
                        }
                    `)
                    if err != nil {
                        log.Error().Err(err).Msg("Failed to execute JavaScript")
                    } else {
                        log.Info().
                            Interface("result", result).
                            Msg("Hot reload completed")
                    }
                }()
            }
        case err, ok := <-watcher.Errors:
            if !ok {
                return nil
            }
            log.Error().Err(err).Msg("Watcher error")
        }
    }
}

Then modify the main function to use development mode:

// At the end of main()
if isDevelopmentMode() {
    log.Info().Msg("Starting file watcher for hot reload")
    if err := watchForChanges(vm); err != nil {
        log.Fatal().Err(err).Msg("Failed to start file watcher")
    }
} else {
    log.Info().Msg("Hot reload disabled in production mode")
}

To run in development mode:

APP_ENV=development go run cmd/server/main.go

Step 11: Build for Production

For production, we want to build a single binary that includes the embedded JavaScript:

go build -o myvm-server cmd/server/main.go

This creates a standalone binary that can be deployed without Node.js or any other dependencies.
